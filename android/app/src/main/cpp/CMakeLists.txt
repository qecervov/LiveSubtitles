cmake_minimum_required(VERSION 3.22)
project(livesubtitles LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_options(-fPIC
    "-ffile-prefix-map=${PRIVATE_PATH_PREFIX}=." "-fdebug-prefix-map=${PRIVATE_PATH_PREFIX}=."
    "-ffile-prefix-map=${NATIVE_ROOT}=native" "-fdebug-prefix-map=${NATIVE_ROOT}=native"
    "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=src" "-fdebug-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=src")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
add_subdirectory(${NATIVE_ROOT}/whisper whisper EXCLUDE_FROM_ALL)
set(WITH_MKL OFF CACHE BOOL "" FORCE)
set(WITH_RUY ON CACHE BOOL "" FORCE)
set(WITH_DNNL OFF CACHE BOOL "" FORCE)
set(WITH_CUDA OFF CACHE BOOL "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_CPU_DISPATCH OFF CACHE BOOL "" FORCE)
set(OPENMP_RUNTIME NONE CACHE STRING "" FORCE)
add_subdirectory(${NATIVE_ROOT}/ctranslate2 ctranslate2 EXCLUDE_FROM_ALL)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(${NATIVE_ROOT}/sentencepiece sentencepiece EXCLUDE_FROM_ALL)
add_library(livesubtitles SHARED native_engine.cpp)
target_include_directories(livesubtitles PRIVATE ${NATIVE_ROOT}/sentencepiece/src)
target_link_libraries(livesubtitles PRIVATE whisper ctranslate2 sentencepiece-static android log)
target_link_options(livesubtitles PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
